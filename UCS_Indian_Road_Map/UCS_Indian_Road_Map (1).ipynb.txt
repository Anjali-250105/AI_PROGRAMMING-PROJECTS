        "# Uniform Cost Search\n",
        "def uniform_cost_search(graph, start, goal):\n",
        "\n",
        "    # Priority queue:\n",
        "    # (total cost, current city, path)\n",
        "    priority_queue = [(0, start, [start])]\n",
        "\n",
        "    visited_cost = {}\n",
        "\n",
        "    while priority_queue:\n",
        "\n",
        "        cost, city, path = heapq.heappop(priority_queue)\n",
        "\n",
        "        # Skip if this city was already reached with lower cost\n",
        "        if city in visited_cost and visited_cost[city] <= cost:\n",
        "            continue\n",
        "\n",
        "        visited_cost[city] = cost\n",
        "\n",
        "        # Goal reached\n",
        "        if city == goal:\n",
        "            return path, cost\n",
        "\n",
        "        # Explore neighbouring cities\n",
        "        for neighbor, distance in graph[city]:\n",
        "\n",
        "            new_cost = cost + distance\n",
        "\n",
        "            new_path = path + [neighbor]\n",
        "\n",
        "            heapq.heappush(\n",
        "                priority_queue,\n",
        "                (new_cost, neighbor, new_path)\n",
        "            )\n",
        "\n",
        "    return None, float(\"inf\")\n",
        "\n",
        "\n",
        "# Starting and goal cities\n",
        "start = \"Hyderabad\"\n",
        "goal = \"Delhi\"\n",
        "\n",
        "\n",
        "# Run UCS\n",
        "path, total_distance = uniform_cost_search(\n",
        "    graph, start, goal\n",
        ")\n",
        "\n",
        "\n",
        "# Display result\n",
        "if path:\n",
        "    print(\"Route:\", \" -> \".join(path))\n",
        "    print(\"Total distance:\", total_distance, \"km\")\n",
        "else:\n",
        "    print(\"No route found.\")"
